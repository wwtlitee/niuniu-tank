 function tank(color,id){
  const base=0xb62632;
  const object=id===1?classicTank(base,base,1):id===5?makeRtsTank(base,base,1):id===6?makeHeroTankModel():makeTank(base,base,1,({2:'light',3:'medium',4:'heavy'})[id]||'tank');
  const turret=object.userData.turret,scale=1.4,turretHeight=turret.position.y*scale;
  object.scale.setScalar(scale);object.rotation.y=Math.PI;object.updateMatrixWorld(true);
  const bodyBuilder=M.builder(),turretBuilder=M.builder(),geometries=new Set(),materials=new Set();
  object.traverse(mesh=>{
   if(!mesh.isMesh)return;
   geometries.add(mesh.geometry);materials.add(mesh.material);
   let node=mesh,isTurret=false;while(node){if(node===turret){isTurret=true;break;}node=node.parent;}
   const geometry=mesh.geometry.clone().applyMatrix4(mesh.matrixWorld);
   if(isTurret)geometry.translate(0,-turretHeight,0);
   const c=mesh.material.color.clone();if(id===1)c.convertLinearToSRGB();
   const painted=c.r>c.g*1.5&&c.r>c.b*1.3;
   (isTurret?turretBuilder:bodyBuilder).add(geometry,painted?color:c.getHex(),0,0,0,0,0,0,painted);
  });
  // Common racing kit: tow points, replaceable track shoes and rear cooling.
  for(const side of [-1,1]){
   const x=side*(id===5?1.38: id===6?1.16:1.06)*scale;
   for(let i=0;i<16;i++)bodyBuilder.box(id===5?.62:.94,.065,.13,0x687477,x,(id===6?.77:.85)*scale,-2.24+i*.29);
   bodyBuilder.ring(.13,.035,0xb7ac86,side*.78,1.05,2.29);
   bodyBuilder.box(.25,.12,.055,0xffdf9b,side*1.05,1.8,2.32);
   bodyBuilder.box(.16,.12,.06,0xed5146,side*1.04,1.5,-2.4);
  }
  for(let i=0;i<7;i++)bodyBuilder.box(.92,.035,.075,0x263139,0,id===5?1.58:1.99,-1.3-i*.1);
  for(const g of geometries)g.dispose();for(const m of materials)m.dispose();
  return {body:bodyBuilder.finish(),turret:turretBuilder.finish(),turretHeight};
 }
 return {tank};
});
